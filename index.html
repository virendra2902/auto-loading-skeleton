Slot System — zero-ambiguity skeletons
import { AutoSkeleton, SkeletonSlot } from 'auto-loading-skeleton-v4';

// Mark explicit regions with SkeletonSlot — confidence = 1.0
function ProductCard({ product }) {
  return (
    <div>
      <SkeletonSlot type="image"   height="200px"   />
      <SkeletonSlot type="heading" width="70%"    />
      <SkeletonSlot type="text"    lines={3}        />
      <SkeletonSlot type="button"  width="120px"  />
      {/* Real content — shown when loading=false */}
      <img src={product.img} />
      <h2>{product.title}</h2>
      <p>{product.desc}</p>
      <button>Add to cart</button>
    </div>
  );
}

useSkeletonStream — React 18 streaming
import { useSkeletonStream, AutoSkeleton } from 'auto-loading-skeleton-v4';

function StreamingFeed() {
  const { chunks, loading, isEmpty } = useSkeletonStream(
    async function*() {
      const stream = await fetch('/api/feed/stream');
      const reader = stream.body.getReader();
      while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        yield JSON.parse(decode(value));
      }
    },
    []
  );

  return (
    <div>
      {chunks.map((chunk, i) => (
        <FeedItem key={i} data={chunk} />
      ))}
      {loading && <AutoSkeleton loading={true} count={3}>
        <FeedItem />
      </AutoSkeleton>}
    </div>
  );
}
